# This entire directory is opt-in. Headless builds never discover Qt/OpenGL.
foreach(toolkit TKV3d TKOpenGl TKService TKMesh)
    if(NOT TARGET ${toolkit})
        message(FATAL_ERROR "GEO-Visualiser requires visualization-enabled OCCT 8.0.1: missing ${toolkit}")
    endif()
endforeach()
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
if(UNIX AND NOT APPLE)
    find_package(X11 REQUIRED)
elseif(NOT WIN32)
    message(FATAL_ERROR "GEO-Visualiser supports Windows and Linux X11/XWayland only")
endif()

add_library(smd-inspector-adapter STATIC model.cpp)
target_link_libraries(smd-inspector-adapter PUBLIC smd-geometry)
target_include_directories(smd-inspector-adapter PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(smd-inspector-adapter PUBLIC cxx_std_20)

add_library(smd-inspector-desktop STATIC viewport.cpp window.cpp)
target_link_libraries(smd-inspector-desktop PUBLIC smd-inspector-adapter Qt6::Widgets TKV3d TKOpenGl TKService TKMesh)
if(UNIX)
    target_link_libraries(smd-inspector-desktop PRIVATE X11::X11)
endif()
add_executable(geo-visualiser main.cpp)
target_link_libraries(geo-visualiser PRIVATE smd-inspector-desktop)
# Resolve the two original fixtures from the checkout, independent of cwd.
configure_file(fixtures.hpp.in generated/fixtures.hpp @ONLY)
target_include_directories(geo-visualiser PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
if(WIN32)
    get_filename_component(smd_qt_prefix "${Qt6_DIR}/../../.." ABSOLUTE)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/start-geo-visualiser.cmd"
        CONTENT "@echo off\nsetlocal\nset \"PATH=$<TARGET_FILE_DIR:TKernel>;$<TARGET_FILE_DIR:Qt6::Core>;%PATH%\"\nset \"QT_QPA_PLATFORM_PLUGIN_PATH=${smd_qt_prefix}/plugins/platforms\"\n\"$<TARGET_FILE:geo-visualiser>\"\n")
endif()

foreach(target smd-inspector-adapter smd-inspector-desktop geo-visualiser)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(BUILD_TESTING)
    add_executable(smd-inspector-tests tests.cpp)
    target_link_libraries(smd-inspector-tests PRIVATE smd-inspector-desktop)
    add_test(NAME inspector-adapter COMMAND smd-inspector-tests "${PROJECT_SOURCE_DIR}/public/samples")
    # Explicit interactive-capable execution, not part of headless CTest.
    # smd-inspector-tests <sample-directory> --render
endif()

cmake_minimum_required(VERSION 3.20)
project(smdviewer_native LANGUAGES CXX)

include(CTest)
option(SMD_BUILD_GEOMETRY "Build GEO-01A native OCCT face conversion" OFF)
option(SMD_BUILD_GEO_VISUALISER "Build the standalone Qt/OCCT geometry inspector" OFF)
if(SMD_BUILD_GEO_VISUALISER AND NOT SMD_BUILD_GEOMETRY)
    message(FATAL_ERROR "SMD_BUILD_GEO_VISUALISER requires SMD_BUILD_GEOMETRY=ON")
endif()
add_subdirectory(libsmd)
if(SMD_BUILD_GEOMETRY)
    if(EMSCRIPTEN)
        message(FATAL_ERROR "GEO-01A requires a native toolchain; OCCT is not part of the WASM build")
    endif()
    add_subdirectory(libgeometry)
endif()
if(SMD_BUILD_GEO_VISUALISER)
    add_subdirectory(geo-visualiser)
endif()
